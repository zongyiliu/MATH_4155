			\item[(i)] Establish the identity
			\[
			\int_{\mathbb{R}} e^{-x^{2}/2}\,dx = \sqrt{2\pi}.
			\]
			
			\item[(ii)] Let the random variable $Z$ have Gaussian $\mathcal{N}(m,\sigma^{2})$ distribution, that is
			\[
			\mathbb{P}(Z \in A)
			=
			\int_{A} (2\pi\sigma^{2})^{-1/2}
			e^{-(x-m)^{2}/(2\sigma^{2})}\,dx,
			\qquad
			A \in \mathcal{B}(\mathbb{R}),
			\]
			for some $m \in \mathbb{R}$ and $\sigma > 0$. Show that
			\[
			\mathbb{E}(Z) = m,
			\qquad
			\mathbb{E}(Z^{2}) = m^{2} + \sigma^{2},
			\qquad
			\operatorname{Var}(Z) = \sigma^{2}.
			\]
			Compute also the Laplace transform
			\[
			\mathbb{E}\!\left(e^{\xi Z}\right)
			=
			e^{m\xi + (\sigma^{2}\xi^{2})/2},
			\qquad
			\xi \in \mathbb{R}.
			\]
			(\emph{Hint}: Reduce the results to those for the standard Gaussian $\mathcal{N}(0,1)$.)
			
			\item[(iii)] Let $X$ be a random variable on a probability space $(\Omega,\mathcal{F},\mathbb{P})$, such that the Laplace transform
			\[
			\phi(\xi) := \mathbb{E}\!\left(e^{\xi X}\right)

		\section{Question 6}
		
		{\color{deepred}\fontspec{Georgia} Convergence in Probability is Metrizable}
		
		
		Let $\mathbb{L}^{0}$ denote the space of (equivalence classes of) random variables
		$X:\Omega\to\mathbb{R}$ on a probability space $(\Omega,\mathcal{F},\mathbb{P})$,
		and define
		\[
		\rho(X,Y):=\mathbb{E}(|X-Y|\wedge 1), \qquad X,Y\in\mathbb{L}^{0}.
		\]
		
		\begin{enumerate}
			\item[(i)] Show that $\rho$ defines a metric on $\mathbb{L}^{0}$, and that convergence under $\rho$ is equivalent to convergence in probability.
			
			\item[(ii)] Show the same holds if $\rho$ is replaced by
			\[
			\varrho(X,Y):=\mathbb{E}\!\left(\frac{|X-Y|}{1+|X-Y|}\right).
			\]
			
			\item[(iii)] (Optional) Does there exist a metric on $\mathbb{L}^{0}$ such that convergence under this metric is equivalent to a.e.\ convergence?
		\end{enumerate}
		
		(See also Problems 4.6 and 4.19 in Walsh (2012))
